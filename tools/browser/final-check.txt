  await page.emulateMedia({reducedMotion:'reduce'});
  await page.goto('http://127.0.0.1:4624/');
  await page.addScriptTag({path:'C:/Users/William King/AppData/Local/npm-cache/_npx/0f94ee7615faf582/node_modules/axe-core/axe.min.js'});
