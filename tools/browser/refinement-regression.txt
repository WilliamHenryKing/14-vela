async (page) => {
  page.setDefaultTimeout(7000);
  await page.emulateMedia({reducedMotion: "no-preference"});
  const results = [];
  const errors = [];
  const assert = (ok, name) => {
    if (!ok) throw new Error(name);
    results.push(name);
  };
  page.on("pageerror", (e) => errors.push(e.message));
  page.on("console", (e) => {
    if (e.type() === "error") errors.push(e.text());
  });
  const capture = async (name) =>
    page.screenshot({ path: `docs/visual/captures/refinement-2026-09-27/${name}.png`, scale: "css" });
  await page.setViewportSize({ width: 1440, height: 1000 });
  await page.goto("http://127.0.0.1:4624/");
  await page.waitForTimeout(2000);
  await capture("hero-desktop");
  await page.getByRole("button", { name: "ink card", exact: true }).click();
  assert(
    await page.locator(".hero .card-front").evaluate((el) => el.classList.contains("finish-ink")),
    "Card finish changes artwork",
  );
  await page.getByRole("button", { name: "Explore Vela", exact: true }).click();
  assert(await page.locator("dialog").evaluate((el) => el.open), "Native dialog opens");
  await page.getByRole("radio", { name: /Plus/ }).check();
  await page.getByRole("button", { name: "See my preview" }).click();
  assert(
    (await page.locator(".demo-summary").innerText()).includes("89"),
    "Selected plan price appears in summary",
  );
  await capture("demo-summary-desktop");
  await page.getByRole("button", { name: "Edit my preview" }).click();
  assert(
    await page.getByRole("radio", { name: /Plus/ }).isChecked(),
    "Preview editing preserves selection",
  );
  await page.keyboard.press("Escape");
  assert((await page.locator("dialog").count()) === 0, "Escape closes dialog");
  assert(
    await page
      .getByRole("button", { name: "Explore Vela", exact: true })
      .evaluate((el) => el === document.activeElement),
    "Dialog restores trigger focus",
  );
  await page.locator("#intro").scrollIntoViewIfNeeded();
  await page.waitForTimeout(1200);
  await capture("statement-desktop");
  await page.getByRole("button", { name: "Control", exact: true }).click();
  await page.waitForTimeout(1200);
  await page.getByRole("switch", { name: "Pause demo card" }).click();
  assert(
    (await page.getByRole("switch").getAttribute("aria-checked")) === "true",
    "Demo card pauses",
  );
  await capture("product-control-desktop");
  await page.getByRole("button", { name: "Organise", exact: true }).click();
  await page.waitForTimeout(900);
  assert((await page.locator(".pocket").count()) === 3, "Organise displays three pockets");
  await page.locator("#goals").scrollIntoViewIfNeeded();
  await page.waitForTimeout(1200);
  await capture("goals-desktop");
  await page.locator("#monthly").fill("0");
  assert(
    (await page.locator(".goal-timing").innerText()).includes("Add a monthly"),
    "Zero monthly contribution explains unreachable goal",
  );
  await page.locator("#starting").fill("50000");
  assert(
    (await page.locator(".goal-timing").innerText()).includes("already reached"),
    "Already-funded goal is explicit",
  );
  await page.locator("#starting").fill("5000");
  await page.locator("#monthly").fill("1500");
  await page.getByRole("button", { name: "Your next chapter", exact: true }).click();
  assert(
    (await page.locator(".goal-timing").innerText()).startsWith("64 months"),
    "Changing goal recomputes months",
  );
  await page.locator("#plans").scrollIntoViewIfNeeded();
  await page.waitForTimeout(1100);
  await capture("plans-desktop");
  await page.locator("summary").filter({ hasText: "How does the savings planner work?" }).click();
  assert((await page.locator("details[open]").count()) === 1, "FAQ disclosure opens");
  await page.locator(".closing").scrollIntoViewIfNeeded();
  await page.waitForTimeout(1100);
  await capture("closing-desktop");
  await page.getByRole("button", { name: "Motion on", exact: true }).click();
  assert(
    (await page.locator(".motion-off").count()) === 1,
    "Manual motion setting reverts animations",
  );
  await page.evaluate(() => window.scrollTo(0, 0));
  await capture("hero-motion-off");
  for (const [width, height] of [
    [1920, 1080],
    [768, 1024],
    [390, 844],
    [320, 740],
    [844, 390],
  ]) {
    await page.setViewportSize({ width, height });
    await page.evaluate(() => window.scrollTo(0, 0));
    await page.waitForTimeout(350);
    const overflow = await page.evaluate(() => document.documentElement.scrollWidth > innerWidth);
    assert(!overflow, `No horizontal overflow at ${width}x${height}`);
    await capture(`hero-${width}x${height}`);
  }
  await page.setViewportSize({ width: 390, height: 844 });
  await page.getByRole("button", { name: "Menu +", exact: true }).click();
  await page.getByRole("navigation", {name:"Main navigation"}).getByRole("link", { name: "Your goals" }).click();
  assert(
    (await page
      .getByRole("button", { name: "Menu +", exact: true })
      .getAttribute("aria-expanded")) === "false",
    "Mobile menu closes after navigation",
  );
  await capture("goals-mobile");
  await page.getByRole("button", { name: "Control", exact: true }).click();
  assert(await page.getByRole("switch").isVisible(), "Mobile product controls remain available");
  await capture("product-mobile");
  await page.getByRole("button", { name: "Explore Plus", exact: true }).click();
  await page.getByRole("button", { name: "See my preview" }).click();
  await capture("demo-mobile");
  await page.keyboard.press("Escape");
  await page.emulateMedia({ reducedMotion: "reduce" });
  await page.goto("http://127.0.0.1:4624/");
  await page.waitForTimeout(350);
  assert(
    (await page.locator(".motion-off").count()) === 1,
    "OS reduced motion respected on cold load",
  );
  assert(await page.locator("h1").isVisible(), "Reduced-motion hero remains visible");
  await page.keyboard.press("Tab");
  assert(
    await page
      .getByRole("link", { name: "Skip to content" })
      .evaluate((el) => el === document.activeElement),
    "Keyboard skip link is first",
  );
  await page.keyboard.press("Enter");
  assert(await page.evaluate(() => location.hash === "#main"), "Skip link navigates to main");
  await capture("reduced-motion-mobile");
  await page.addScriptTag({
    path: "C:/Users/William King/AppData/Local/npm-cache/_npx/0f94ee7615faf582/node_modules/axe-core/axe.min.js",
  });
  const accessibility = await page.evaluate(async () => {
    const r = await axe.run(document, {
      runOnly: { type: "tag", values: ["wcag2a", "wcag2aa", "wcag21aa"] },
    });
    return r.violations.map((v) => ({
      id: v.id,
      impact: v.impact,
      nodes: v.nodes.map((n) => ({ target: n.target, summary: n.failureSummary })),
    }));
  });
  assert(errors.length === 0, "No browser console errors");
  return JSON.stringify(
      {
        date: new Date().toISOString(),
        browser: await page.evaluate(() => navigator.userAgent),
        devicePixelRatio: await page.evaluate(() => devicePixelRatio),
        checks: results,
        errors,
        accessibility,
        scope:
          "Functional automation and sampled viewport captures; not continuous motion viewing or physical-phone evidence.",
      },
      null,
      2,
  );
}
